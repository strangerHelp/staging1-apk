            AndroidView(
                modifier = Modifier.fillMaxSize(),
                factory = { ctx ->
                    MapView(ctx).apply {
                        setTileSource(TileSourceFactory.MAPNIK)
                        setMultiTouchControls(true)
                        controller.setZoom(13.0)
                        controller.setCenter(centerPoint)
                        
                        // Add pins
                        pins.forEach { pin ->
                            val marker = Marker(this)
                            marker.position = pin.location
                            marker.setAnchor(Marker.ANCHOR_CENTER, Marker.ANCHOR_BOTTOM)
                            // We can use a simple icon or just default
                            val drawableId = if (pin.isHelper) android.R.drawable.ic_menu_myplaces else android.R.drawable.ic_menu_info_details
                            marker.icon = androidx.core.content.ContextCompat.getDrawable(ctx, drawableId)
                            marker.title = if (pin.isHelper) "Helper Online" else "Open Task"
                            overlays.add(marker)
                        }
                    }
                },
                update = { view -> 
                    // Update if needed
                }
            )
